-- PING CALCULATOR (ACCURATE)
-- ══════════════════════════════════════════════════════════════
local function GetPing()
    local PlayerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not PlayerGui then return 0.15 end

    local NetworkStats = PlayerGui:FindFirstChild("NetworkStats")
    if not NetworkStats then return 0.15 end

    local PingLabel = NetworkStats:FindFirstChild("PingLabel")
    if not PingLabel then return 0.15 end

    local PingText = PingLabel.Text
    if typeof(PingText) ~= "string" then return 0.15 end

    local PingValue = tonumber(PingText:match("%d+"))
    if not PingValue then return 0.15 end

    local PingSeconds = PingValue / 1000
    return math.clamp(PingSeconds, 0.01, 2)
end

-- ══════════════════════════════════════════════════════════════
-- VELOCITY TRACKER (HISTORY-BASED PREDICTION)
-- ══════════════════════════════════════════════════════════════
local VelocityHistory = {}
local MaxHistoryFrames = 8

RunService.Heartbeat:Connect(function()
    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.Character then
            local RootPart = Player.Character:FindFirstChild("HumanoidRootPart")
            if RootPart then
                if not VelocityHistory[Player] then
                    VelocityHistory[Player] = {}
                end
                table.insert(VelocityHistory[Player], {
                    Time = os.clock(),
                    Position = RootPart.Position,
                    Velocity = RootPart.AssemblyLinearVelocity
                })
                if #VelocityHistory[Player] > MaxHistoryFrames then
                    table.remove(VelocityHistory[Player], 1)
                end
            else
                VelocityHistory[Player] = nil
            end
        end
    end
end)

Players.PlayerRemoving:Connect(function(Player)
    VelocityHistory[Player] = nil
end)

local function CalculateVelocity(Player)
    local History = VelocityHistory[Player]
    if not History or #History < 2 then
        return Vector3.new(0, 0, 0)
    end

    local TotalVelocity = Vector3.new(0, 0, 0)
    local ValidSamples = 0

    for i = 2, #History do
        local TimeDelta = History[i].Time - History[i-1].Time
        if TimeDelta > 0 then
            local PositionDelta = History[i].Position - History[i-1].Position
            local InstantVelocity = PositionDelta / TimeDelta
            TotalVelocity = TotalVelocity + InstantVelocity
            ValidSamples = ValidSamples + 1
        end
    end

    if ValidSamples == 0 then return Vector3.new(0, 0, 0) end

    local AvgVelocity = TotalVelocity / ValidSamples

    -- Clamp extreme Y velocities (anti-cheat evasion)
    if AvgVelocity.Y > 200 then
        return Vector3.new(AvgVelocity.X * 1.2, math.clamp(AvgVelocity.Y * 0.8, 0, 500), AvgVelocity.Z * 1.2)
    end

    return AvgVelocity
end

-- ══════════════════════════════════════════════════════════════
-- GOD-TIER PREDICTION SYSTEM
-- ══════════════════════════════════════════════════════════════
local function PredictPosition(TargetPart, TargetRootPart)
    if not TargetPart then return Vector3.new(0, 0, 0) end

    local Ping = GetPing()
    local Velocity = CalculateVelocity(Players:GetPlayerFromCharacter(TargetPart.Parent))

    -- Multi-layer prediction
    local BasePrediction = TargetPart.Position + (Velocity * Ping * AimbotSettings.PredictionMultiplier)

    -- Gravity compensation for projectiles
    local GravityDrop = Vector3.new(0, 4.9 * Ping * Ping, 0)
    local FinalPosition = BasePrediction + GravityDrop

    return FinalPosition
end

local function IsPlayerExcluded(PlayerName)
    for _, Excluded in ipairs(ExcludedPlayers) do
        if Excluded ~= "" and string.find(string.lower(PlayerName), string.lower(Excluded)) then
            return true
        end
    end
    return false
end

local function GetClosestTarget()
    local ClosestTarget = nil
    local ClosestDistance = AimbotSettings.FOV
    local ScreenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.Character then
            local TargetPart = Player.Character:FindFirstChild(AimbotSettings.SelectedPart)
            local HumanoidCheck = Player.Character:FindFirstChild("Humanoid")
            local RootPart = Player.Character:FindFirstChild("HumanoidRootPart")

            if TargetPart and HumanoidCheck and HumanoidCheck.Health > 0 and RootPart then
                -- Check exclusion
                if IsPlayerExcluded(Player.Name) then continue end

                -- Check distance
                local LocalRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if LocalRoot then
                    local Distance3D = (TargetPart.Position - LocalRoot.Position).Magnitude
                    if Distance3D < AimbotSettings.MinDistance then continue end
                    if Distance3D > AimbotSettings.MaxDistance then continue end
                end

                -- Wall check
                if AimbotSettings.WallCheck and LocalRoot then
                    local Direction = TargetPart.Position - LocalRoot.Position
                    if Direction.Magnitude < 1 then continue end

                    local RaycastResult = Workspace:Raycast(LocalRoot.Position, Direction, RaycastParams.new())
                    if RaycastResult then
                        local HitPart = RaycastResult.Instance
                        if HitPart and HitPart.Parent ~= Player.Character then
                            continue
                        end
                    end
                end

                -- Screen check
                local ScreenPosition, OnScreen = Camera:WorldToViewportPoint(TargetPart.Position)
                if not OnScreen or ScreenPosition.Z <= 0 then continue end

                local ScreenPos2D = Vector2.new(ScreenPosition.X, ScreenPosition.Y)
                local Distance2D = (ScreenPos2D - ScreenCenter).Magnitude

                if Distance2D <= ClosestDistance then
                    ClosestDistance = Distance2D
                    ClosestTarget = Player
                end
            end
        end
    end

    return ClosestTarget
end

    if not Char then return nil end

    for _, Child in ipairs(Char:GetChildren()) do
        if pcall(function() return Child:IsA("Tool") end) and Child:IsA("Tool") then
            return Child

        for _, Child in ipairs(Backpack:GetChildren()) do
            if pcall(function() return Child:IsA("Tool") end) and Child:IsA("Tool") then

            end
        end
    end

    return nil
end

-- ══════════════════════════════════════════════════════════════
local AutoAttackEnabled = false
local AutoAttackRunning = false
local AttackCooldown = 0.35

local function GetPlayersInRange(Range)
    local PlayersInRange = {}
    local LocalChar = LocalPlayer.Character
    if not LocalChar or not LocalChar.PrimaryPart then return PlayersInRange end

    local LocalPos = LocalChar.PrimaryPart.Position

    for _, Player in pairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.Character and Player.Character.PrimaryPart then
            local Success, Distance = pcall(function()
                return (Player.Character.PrimaryPart.Position - LocalPos).Magnitude
            end)
            if Success and Distance and Distance <= Range then
                table.insert(PlayersInRange, Player)
            end
        end
    end
    return PlayersInRange
end

local function PerformMeleeAttack()
    if not SendRemote then return end

    local Tool = GetActiveTool()
    if not Tool or not IsMelee(Tool) then return end

    local NearbyPlayers = GetPlayersInRange(25)
    if #NearbyPlayers == 0 then return end

    local LocalChar = LocalPlayer.Character
    if not LocalChar or not LocalChar.PrimaryPart then return end

    local LocalPos = LocalChar.PrimaryPart.Position
    local Targets = {}
    local TargetPositions = {}

    for _, Player in pairs(NearbyPlayers) do
        if Player.Character then
            local Head = Player.Character:FindFirstChild("Head")
            local Root = Player.Character.PrimaryPart
            if Head and Root then
                local PredPos = PredictPosition(Head, Root)
                table.insert(Targets, Player)
                table.insert(TargetPositions, PredPos)
            end
        end
    end

    if #Targets == 0 then return end

    local AimDirection = CFrame.lookAt(LocalPos, TargetPositions[1])

    local AttackArgs = {
        "melee_attack",
        Tool,
        Targets,
        AimDirection,
        0.8
    }

    pcall(function()
        Counter = Counter + 1
        SendRemote:FireServer(Counter, unpack(AttackArgs))
    end)
end

local function StartAutoAttack()
    if AutoAttackRunning then return end
    AutoAttackRunning = true

    task.spawn(function()
        while AutoAttackEnabled do
            task.wait(AttackCooldown)
            if LocalPlayer.Character and LocalPlayer.Character.PrimaryPart then
                pcall(PerformMeleeAttack)
            end
        end
        AutoAttackRunning = false
    end)
end

local MeleeAuraEnabled = false
local OriginalFistAttributes = {}

local function ModifyFists(Tool, Enable)
    if not Tool then return end

    local Attributes = Tool:GetAttributes()
    local AttrKeys = {}
    for Key in pairs(Attributes) do
        table.insert(AttrKeys, Key)
    end
    table.sort(AttrKeys)

    if #AttrKeys >= 7 then
        local RangeKey = AttrKeys[6]
        local DamageKey = AttrKeys[7]

        if Enable then
            if OriginalFistAttributes[RangeKey] == nil then
                OriginalFistAttributes[RangeKey] = Tool:GetAttribute(RangeKey)
            end
            if OriginalFistAttributes[DamageKey] == nil then
                OriginalFistAttributes[DamageKey] = Tool:GetAttribute(DamageKey)
            end
            Tool:SetAttribute(RangeKey, 500)
            Tool:SetAttribute(DamageKey, 999)
        else
            if OriginalFistAttributes[RangeKey] then
                Tool:SetAttribute(RangeKey, OriginalFistAttributes[RangeKey])
            end
            if OriginalFistAttributes[DamageKey] then
                Tool:SetAttribute(DamageKey, OriginalFistAttributes[DamageKey])
            end
        end
    end
end

local function CheckAndModifyFists()
    local Char = LocalPlayer.Character
    local Backpack = LocalPlayer:FindFirstChild("Backpack")
    if not Char or not Backpack then return end

    local Tools = {}
    for _, Child in ipairs(Char:GetChildren()) do
        if IsMelee(Child) then table.insert(Tools, Child) end
    end
    for _, Child in ipairs(Backpack:GetChildren()) do
        if IsMelee(Child) then table.insert(Tools, Child) end
    end

    for _, Tool in ipairs(Tools) do
        ModifyFists(Tool, MeleeAuraEnabled)
    end
end

local function LoadESP()
    for _, Player in pairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and not ESPPlayers[Player] then
            CreateESP(Player)
        end
    end

    Players.PlayerAdded:Connect(function(Player)
        if Player ~= LocalPlayer then
            Player.CharacterAdded:Connect(function()
                task.wait(0.1)
                if not ESPPlayers[Player] then
                    CreateESP(Player)
                end
            end)
            if Player.Character and not ESPPlayers[Player] then
                task.wait(0.1)
                CreateESP(Player)
            end
        end
    end)

    Players.PlayerRemoving:Connect(function(Player)
        if ESPPlayers[Player] then
            for _, Drawing in pairs(ESPPlayers[Player].Drawings) do
                if Drawing and Drawing.Destroy then
                    pcall(function() Drawing:Destroy() end)
                end
            end
            if ESPPlayers[Player].Connection then
                pcall(function() ESPPlayers[Player].Connection:Disconnect() end)
            end
            ESPPlayers[Player] = nil
        end
    end)
end

local function IsDowned()
    local Hum = GetHumanoid()
    if not Hum then return false end
    if Hum.Health <= 0 then return false end
    return Hum:GetAttribute("HasBeenDowned") or Hum:GetAttribute("IsDead")
end

local function GetRootPart()
    local Modules = ReplicatedStorage:WaitForChild("Modules")
    local CharModule = Modules:FindFirstChild("Core") and require(Modules.Core.Char)

        local Char = CharModule.current_char.get()
        if Char then return Char:FindFirstChild("HumanoidRootPart") end
    end
    return HumanoidRootPart
end

local function TeleportUnderground()
    local Root = GetRootPart()
    if not Root then return end
    UndergroundPosition = Root.CFrame + Vector3.new(0, -60, 0)
    Root.CFrame = UndergroundPosition
end

local function FlickerAndMove()
    if IsFlickering then return end
    IsFlickering = true

    task.spawn(function()
        while IsFlickering and AntiKillEnabled and IsDowned() do
            local Hum = GetHumanoid()
            if Hum and Hum.Health <= 0 then break end

            local Root = GetRootPart()
            if Root and UndergroundPosition then
                local Angle = math.random() * math.pi * 2
                local Offset = Vector3.new(math.cos(Angle), 0, math.sin(Angle)) * 35
                local NewPos = UndergroundPosition.Position + Offset
                Root.CFrame = CFrame.new(NewPos)
                task.wait(0.06)
                Root.CFrame = UndergroundPosition
            end

            task.wait(0.12)
        end
        IsFlickering = false
    end)
end

RunService.Heartbeat:Connect(function()
    if not AntiKillEnabled then return end

    if IsDowned() then
        local Root = GetRootPart()
        if Root and not UndergroundPosition then
            TeleportUnderground()
               end
        FlickerAndMove()
    else
        if UndergroundPosition then
            local Root = GetRootPart()
            if Root then
                Root.CFrame = UndergroundPosition + Vector3.new(0, 60, 0)
            end
        end
        UndergroundPosition = nil
        IsFlickering = false
    end
end)

-- AUTO FINISH SYSTEM

local AutoFinishEnabled = false
local FinishHoldTime = 1

local function SetFinishPrompt(Prompt)
    if Prompt and Prompt:IsA("ProximityPrompt") then
        Prompt.HoldDuration = FinishHoldTime
        Prompt.MaxActivationDistance = 25
    end
end

local function TryHoldPrompt(Prompt, HoldTime)
    if not Prompt or Prompt:GetAttribute("__AutoFinishBusy") then return end
    Prompt:SetAttribute("__AutoFinishBusy", true)

    pcall(function() if Prompt.InputHoldBegin then Prompt:InputHoldBegin() end end)
    pcall(function() if Prompt.HoldBegin then Prompt:HoldBegin() end end)
    pcall(function() if Prompt.Trigger then Prompt:Trigger() end end)

    task.wait(HoldTime)

    pcall(function() if Prompt.InputHoldEnd then Prompt:InputHoldEnd() end end)
    pcall(function() if Prompt.HoldEnd then Prompt:HoldEnd() end end)

    Prompt:SetAttribute("__AutoFinishBusy", nil)
end

local function FindFinishPrompts()
    local Prompts = {}
    for _, Model in pairs(Workspace:GetChildren()) do
        local PlayerFromChar = Players:GetPlayerFromCharacter(Model)
        if PlayerFromChar and not IsPlayerExcluded(PlayerFromChar.Name) then
            local Root = Model:FindFirstChild("HumanoidRootPart")
            if Root then
                local Prompt = Root:FindFirstChild("FinishPrompt")
                if Prompt then
                    SetFinishPrompt(Prompt)
                    table.insert(Prompts, Prompt)
                end
            end
        end
    end
    return Prompts
end

task.spawn(function()
    while true do
        task.wait(0.3)
        if AutoFinishEnabled then
            for _, Prompt in ipairs(FindFinishPrompts()) do
                task.spawn(TryHoldPrompt, Prompt, FinishHoldTime)
            end
        end
    end
end)

local CrateSkipEnabled = false

local function TrySkipCrate()
    local Success, CrateModule = pcall(function()
        return require(ReplicatedStorage.Modules.Game.CrateSystem.Crate)
    end)

    if not (Success and CrateModule) then return end

    task.spawn(function()
        local SpinningState = CrateModule.spinning
        if not SpinningState then return end

        local WaitTime = 0
        while not SpinningState.get() do
            if WaitTime > 3 then break end
            task.wait(0.05)
            WaitTime = WaitTime + 0.05
        end

        if SpinningState.get() then
            pcall(function()
                CrateModule.skip_spin()
            end)
        end
    end)
end

local function SetupCrateSkip()
    local Remotes = ReplicatedStorage:WaitForChild("Remotes", 5)
    if not Remotes then return end

    local SendEvent = Remotes:WaitForChild("Send", 5)
    if not (SendEvent and SendEvent:IsA("RemoteEvent")) then return end

    SendEvent.OnClientEvent:Connect(function(...)
        if CrateSkipEnabled then
            TrySkipCrate()
        end
    end)
end

SetupCrateSkip()

ReplicatedStorage.ChildAdded:Connect(function(Child)
    if Child.Name == "Remotes" then
        SetupCrateSkip()
    end
end)

            if AimbotSettings.SilentAim or AimbotSettings.RedLineLock then
                FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

            local TargetPart = TargetChar:FindFirstChild(AimbotSettings.SelectedPart)

            if TargetHumanoid and TargetHumanoid.Health > 0 and TargetPart then
                local PredictedPos = PredictPosition(TargetPart, TargetChar:FindFirstChild("HumanoidRootPart"))
                AimLockPosition = AimLockPosition:Lerp(PredictedPos, AimbotSettings.Smoothness)
                SmoothAimPosition = AimLockPosition

                local ScreenPos, OnScreen = Camera:WorldToViewportPoint(SmoothAimPosition)
                if OnScreen then
                    local ScreenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                    local TargetScreenPos = Vector2.new(ScreenPos.X, ScreenPos.Y)

                    if AimbotSettings.RedLineLock then
                        RedLine.Visible = true
                        RedLine.From = ScreenCenter
                        RedLine.To = TargetScreenPos
                        RedLine.Color = Color3.fromRGB(255, 50, 50)
                        RedLine.Thickness = 1.5
                    end

                    local HeadPos, HeadOnScreen = Camera:WorldToViewportPoint(TargetPart.Position + Vector3.new(0, 0.6, 0))
                    local FootPos, FootOnScreen = Camera:WorldToViewportPoint(TargetPart.Position - Vector3.new(0, 0.6, 0))

                    if HeadOnScreen and FootOnScreen then
                        local BoxSize = math.clamp((HeadPos - FootPos).Magnitude / 2, 8, 25)
                        local HalfBox = BoxSize

                        local TopLeft = Vector2.new(ScreenPos.X - HalfBox, ScreenPos.Y - BoxSize)
                        local TopRight = Vector2.new(ScreenPos.X + HalfBox, ScreenPos.Y - BoxSize)
                        local BottomLeft = Vector2.new(ScreenPos.X - HalfBox, ScreenPos.Y + BoxSize)
                        local BottomRight = Vector2.new(ScreenPos.X + HalfBox, ScreenPos.Y + BoxSize)

                        TracerLines[1].From, TracerLines[1].To = TopLeft, TopRight
                        TracerLines[2].From, TracerLines[2].To = TopRight, BottomRight
                        TracerLines[3].From, TracerLines[3].To = BottomRight, BottomLeft
                        TracerLines[4].From, TracerLines[4].To = BottomLeft, TopLeft

                            TracerLines[i].Visible = true
                        end

                    for i = 1, 4 do TracerLines[i].Visible = false end
                end
            else
                RedLine.Visible = false
                for i = 1, 4 do TracerLines[i].Visible = false end
            end
        else
            RedLine.Visible = false
            for i = 1, 4 do TracerLines[i].Visible = false end
            SmoothAimPosition = Vector3.new(0, 0, 0)

            local Tool = GetActiveTool()
            if Tool and IsGun(Tool) then
                mouse1click()

            end
        end

MiscTab:Button({
    Title = "Server Hop (Random)",
    Desc = "Hop to a random server",
    Callback = function()
        local HttpService = game:GetService("HttpService")
        local TeleportService = game:GetService("TeleportService")

        local Success, Response = pcall(function()
            return HttpService:JSONDecode(
                game:HttpGet(
                    "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Desc&limit=100"
                )
            )
        end)

        if not Success or not Response or not Response.data then return end

        local AvailableServers = {}
        for _, Server in ipairs(Response.data) do
            if Server.playing < Server.maxPlayers and Server.id ~= game.JobId then
                table.insert(AvailableServers, Server)
            end
        end

        if #AvailableServers == 0 then return end

        table.sort(AvailableServers, function(a, b) return a.playing > b.playing end)

        local TargetServer = AvailableServers[1]

        TeleportService:TeleportToPlaceInstance(PlaceId, TargetServer.id, LocalPlayer)
    end
})
